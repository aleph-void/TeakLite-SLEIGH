# Hand-computed expectations for the Teak module.  Accumulators are 40 bit (10 hex digits), flags are 0/1.  The words
# were worked out from the instruction encodings (GBATEK / the decoder table) and the expectations by hand from the
# semantics in GBATEK; nothing is copied from the module's output.  "asm" guards each encoding against typos.
defaults SATA 1 CMD 1 CPC 1 CCNTA 1 CREP 1

case mov_imm_positive
asm mov 0x1234,b0
code 5E20 1234
expect b0 1234
expect Z 0
expect M 0
expect E 0
expect N 0
end

case mov_imm_negative
asm mov 0x8000,b0
code 5E20 8000
expect b0 FFFFFF8000
expect M 1
expect Z 0
expect E 0
expect N 0
end
